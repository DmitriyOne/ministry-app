> **For agentic workers:** REQUIRED SUB-SKILL: Use
> superpowers:subagent-driven-development (recommended) or

**Goal:** открыть приложение людям с выданными учётными данными — вход по
email и паролю, роль в JWT, нижняя навигация и пять новых экранов.

**Architecture:** Supabase Auth с выключенной регистрацией; роль
приезжает в JWT через `custom_access_token_hook`; доступ решает
`proxy.ts` по `getClaims()`, заменяя удалённый `LAUNCH_MODE`; отказы
бэкенда показываются тостом, а не ошибкой поля и не 500.

**Tech Stack:** Next.js 16 (App Router, Turbopack), `@supabase/ssr`,
Supabase CLI 2.117.0, zod 4, react-hook-form, next-intl, Tailwind 4,
react-aria-components, pgTAP, vitest.

**Spec:** `docs/superpowers/specs/2026-10-05-v2-auth-menu-design.md`

## Review Focus

Входы и состояния, которые спека подразумевает, но ни один тест задач
прямо не проверяет. Для каждого добавлен тест в задачу-владельца.

1. **Вход при недоступном Supabase** не должен выглядеть как «неверный
   пароль»: `AuthRetryableFetchError` обязан давать `"unavailable"` и
   тост, а не ошибку под полем. Тест — задача 7.
2. **Человек без строки в `user_roles`** (роль удалили руками в Studio):
   хук не должен ломать вход, claims приезжают без `user_role`, разбор
   не падает, а отдаёт роль по умолчанию. Тесты — задачи 2 и 3.
3. **Истёкший токен при недоступном Auth-сервере** уводит на `/login` —
   проверить, что это именно redirect, а не 500 и не бесконечный цикл
   `/login → /day → /login`. Тест — задача 9.
4. **`sessionStorage` бросает** (приватный режим Safari): приветственный
   тост не должен ронять `/day`. Тест — задача 10.
5. **Повторная отправка формы сброса** с того же IP: лимитер обязан
   отвечать так же нейтрально, как успех, не раскрывая ни лимита, ни
   существования аккаунта. Тест — задача 11.

    where id = '11111111-1111-1111-1111-111111111111'),
  now(),
  'the trigger sets updated_at to the current transaction timestamp'
);

`backend/supabase/migrations/20261005120000_profiles_and_roles.sql`:

```sql
-- Profiles and roles for v2 (ADR 0043). This is the first table in this
-- project with real RLS policies: the v1 rule — close the table
-- completely and expose one RPC (ADR 0033/0037) — was derived from a
-- case where an anonymous visitor writes rows it must never read. Here
-- a row belongs to the person reading it, under their own JWT, so a
-- policy is the direct tool and an RPC wrapper would be a layer to
-- write and test for no gain in safety.
--
-- user_roles is the exception to the exception: it keeps the v1 model
-- with no policy at all. A person learns their role from their own JWT
-- (public.custom_access_token_hook, the next migration), never by
-- reading this table.

create type public.app_role as enum ('user', 'admin');

create table public.profiles (
  id uuid primary key references auth.users on delete cascade,
  display_name text not null check (length(trim(display_name)) > 0),
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

create table public.user_roles (
  user_id uuid primary key references auth.users on delete cascade,
  role public.app_role not null default 'user'
);

alter table public.profiles enable row level security;
alter table public.user_roles enable row level security;

-- Grants are issued to a new table in public automatically and policies
-- do not take them back, so they are revoked explicitly — the same
-- reasoning as the v1 migrations.
revoke all on table public.profiles from anon, authenticated;
revoke all on table public.user_roles from anon, authenticated;

-- A column grant, not a table grant. A policy restricts which ROWS may
-- be changed and says nothing about which COLUMNS: with `grant update on
-- table`, a client could run `update profiles set created_at = ...` and
-- rewrite its own timestamps. display_name is the only field a person
-- may change, so it is the only one granted.
grant select on table public.profiles to authenticated;
grant update (display_name) on table public.profiles to authenticated;
grant select, insert, update, delete on table public.profiles to service_role;
grant select, insert, update, delete on table public.user_roles to service_role;

-- (select auth.uid()), not auth.uid(): wrapped in a subquery the planner
-- evaluates it once per statement instead of once per row.
create policy "read own profile" on public.profiles
  for select to authenticated using ((select auth.uid()) = id);

create policy "update own profile" on public.profiles
  for update to authenticated using ((select auth.uid()) = id)
  with check ((select auth.uid()) = id);

-- No insert and no delete policy on purpose: accounts are issued by the
-- owner, so a person can neither create a profile for themselves nor
-- remove one.

-- Table-agnostic on purpose: every table that gets an updated_at column
-- reuses this one trigger function. security invoker, unlike
-- handle_new_user below — it only assigns to NEW and needs no rights the
-- caller has not got. now() resolves under an empty search_path because
-- pg_catalog is always searched implicitly; anything outside it would
-- have to be written out in full.
--
-- profiles is the first table here a client updates directly rather than
-- through an RPC, which is what makes the trigger necessary: there is no
-- single write path that could set updated_at itself, and the column
-- grant above deliberately denies the client any way to set it by hand.
create or replace function public.set_updated_at()
returns trigger
language plpgsql
security invoker
set search_path = ''
as $$
begin
  new.updated_at := now();
  return new;
end;
$$;

create trigger profiles_set_updated_at
  before update on public.profiles
  for each row execute function public.set_updated_at();

-- security definer with an empty search_path: the function needs rights
-- on public tables that the caller (the auth schema's insert) has not
-- got, and an empty search_path means every name below is written out
-- in full.
create or replace function public.handle_new_user()
returns trigger
language plpgsql
security definer
set search_path = ''
as $$
begin
  insert into public.profiles (id, display_name)
  values (
    new.id,
    coalesce(
      nullif(trim(new.raw_user_meta_data ->> 'display_name'), ''),
      split_part(new.email, '@', 1)
    )
  );

  insert into public.user_roles (user_id, role) values (new.id, 'user');

  return new;
end;
$$;

create trigger on_auth_user_created
  after insert on auth.users
  for each row execute function public.handle_new_user();
```

`backend/supabase/migrations/20261005130000_custom_access_token_hook.sql`:

```sql
-- Puts the person's role into their access token (ADR 0044), so the
-- front end and proxy.ts read it from the JWT instead of querying a
-- table on every request.
--
-- The second half of Supabase's RBAC example — a role_permissions table
-- and an authorize(permission) function — is deliberately NOT here:
-- with two roles and no data tables there is nothing to authorise
-- against. It returns with the first ministry tables.
--
-- Enabling the hook is a PROJECT SETTING, not part of a migration:
-- [auth.hook.custom_access_token] in config.toml locally, and
-- Authentication → Hooks in the cloud dashboard for production.
create or replace function public.custom_access_token_hook(event jsonb)
returns jsonb
language plpgsql
stable
security definer
set search_path = ''
as $$
declare
  claims jsonb;
  found_role public.app_role;
begin
  claims := event -> 'claims';

  select role into found_role
    from public.user_roles
    where user_id = (event ->> 'user_id')::uuid;

  -- No row means the role was removed by hand. Signing in must still
  -- work, so the claims go back untouched and the front end applies its
  -- own default (shared/lib/auth).
  if found_role is not null then
    claims := jsonb_set(claims, '{user_role}', to_jsonb(found_role::text));
  end if;

  return jsonb_build_object('claims', claims);
end;
$$;

grant usage on schema public to supabase_auth_admin;
grant execute on function public.custom_access_token_hook(jsonb) to supabase_auth_admin;
revoke execute on function public.custom_access_token_hook(jsonb)
  from public, anon, authenticated;

-- The hook reads user_roles, which has no policies and no grants for the
-- PostgREST roles; the auth admin needs its own explicit grant.
grant select on table public.user_roles to supabase_auth_admin;
```

```ts
import { z } from "zod";

// Mirrors public.app_role (ADR 0043). Kept here rather than imported
// from the generated database types: proxy.ts and client components both
// read a role, and neither should pull in the whole schema typing for it.
export const APP_ROLES = ["user", "admin"] as const;

export type AppRole = (typeof APP_ROLES)[number];

// The least privileged role, applied whenever the claim is missing or
// unrecognised — a person whose user_roles row was deleted by hand can
// still sign in (the hook returns claims untouched), and must not end up
// with more rights than the database would give them.
export const DEFAULT_APP_ROLE: AppRole = "user";

export type SessionClaims = {
  userId: string;
  role: AppRole;
};

// .catch(), not .optional(): an unknown role string is as uninteresting
// as a missing one — both mean "we do not know", and both resolve to the
// default rather than failing the request.
const claimsSchema = z.object({
  sub: z.string().min(1),
  user_role: z.enum(APP_ROLES).catch(DEFAULT_APP_ROLE).default(DEFAULT_APP_ROLE),
});

export const parseSessionClaims = (claims: unknown): SessionClaims | null => {
  const result = claimsSchema.safeParse(claims);

  if (!result.success) return null;

  return { userId: result.data.sub, role: result.data.user_role };
};
```

```tsx
"use client";

import { createContext, useCallback, useMemo, useRef, useState } from "react";
import type { ReactNode } from "react";

import { ToastRegion, TOAST_CONFIG, type QueuedToast, type ToastInput } from "./toast";

type ToastContextValue = {
  showToast: (toast: ToastInput) => void;
};

export const ToastContext = createContext<ToastContextValue | null>(null);

type ToastProviderProps = {
  children: ReactNode;
};

// One region for the whole app, so a feature only has to say what to
// show. The visual and the exit behaviour follow
// features/network-status/ui/ConnectionRestoredToast.tsx, which is left
// alone on purpose (ADR 0046): it works, and moving it here is a
// separate task rather than a quiet widening of v2.
export const ToastProvider = ({ children }: ToastProviderProps) => {
  const [toasts, setToasts] = useState<QueuedToast[]>([]);
  const nextId = useRef(0);

  const remove = useCallback((id: number) => {
    setToasts((current) => current.filter((toast) => toast.id !== id));
  }, []);

  const showToast = useCallback((toast: ToastInput) => {
    const id = nextId.current;
    nextId.current += 1;

    setToasts((current) => [...current, { ...toast, id, isLeaving: false }]);

    window.setTimeout(() => {
      setToasts((current) =>
        current.map((queued) =>
          queued.id === id ? { ...queued, isLeaving: true } : queued,
        ),
      );
      window.setTimeout(() => remove(id), TOAST_CONFIG.exitFallbackMs);
    }, TOAST_CONFIG.visibleMs);
  }, [remove]);

  const value = useMemo(() => ({ showToast }), [showToast]);

  return (
    <ToastContext.Provider value={value}>
      {children}
      <ToastRegion toasts={toasts} onExited={remove} />
    </ToastContext.Provider>
  );
};
```

```ts
import { AuthApiError, AuthRetryableFetchError } from "@supabase/supabase-js";

export type AuthFailure = "invalid-credentials" | "unavailable";

// Which of the two the person is looking at decides where it is shown:
// a credentials problem goes under the field, an unavailable backend
// goes into a toast (ADR 0046). Guessing is not needed — Supabase is
// specific: 400 with invalid_credentials for a wrong password,
// AuthRetryableFetchError or a 5xx when it could not be reached.
//
// Anything unrecognised is treated as unavailable rather than as bad
// credentials: telling someone their password is wrong when it is not
// sends them round in circles.
export const classifyAuthError = (error: unknown): AuthFailure => {
  if (error instanceof AuthRetryableFetchError) return "unavailable";

  if (error instanceof AuthApiError) {
    if (error.status === 400 && error.code === "invalid_credentials") {
      return "invalid-credentials";
    }
  }

  return "unavailable";
};
```

```ts
import { routes } from "@/shared/config/routes";

// Paths that need a session. /soon, /install, /about and /privacy are
// deliberately absent: they are the public side of the app (ADR 0045).
const SESSION_REQUIRED_PATHS: string[] = [
  routes.root,
  routes.day,
  routes.calendar,
  routes.month,
  routes.year,
  routes.profile,
];

// Paths that only make sense without a session. A signed-in person
// landing here goes to the app instead — and nothing in this table may
// point back at a path in it, or the two redirects form a loop.
const SESSION_FORBIDDEN_PATHS: string[] = [routes.login, routes.forgotPassword];

type GateInput = {
  pathname: string;
  hasSession: boolean;
};

export type GateDecision =
  | { action: "pass" }
  | { action: "redirect"; to: string };

// The whole access decision for v2, as a pure function so the redirect
// loop between /login and /day is caught by a test rather than by a
// person stuck in a browser. proxy.ts only reads the cookies and applies
// what this returns.
export const resolveGate = ({ pathname, hasSession }: GateInput): GateDecision => {
  if (SESSION_FORBIDDEN_PATHS.includes(pathname)) {
    return hasSession ? { action: "redirect", to: routes.day } : { action: "pass" };
  }

  if (SESSION_REQUIRED_PATHS.includes(pathname) && !hasSession) {
    return { action: "redirect", to: routes.login };
  }

  return { action: "pass" };
};
```
